## Technical Details

### OpenTelemetry Setup (server/tracing.ts)

```typescript
import { BasicTracerProvider, ConsoleSpanExporter, SimpleSpanProcessor, BatchSpanProcessor } from '@opentelemetry/sdk-trace-node';
import { JaegerExporter } from '@opentelemetry/exporter-jaeger-basic';
import { registerInstrumentations } from '@opentelemetry/auto-instrumentations-node';
import { NodeSDK } from '@opentelemetry/sdk-node';

export function initializeTracing() {
  const jaegerExporter = new JaegerExporter({
    host: process.env.JAEGER_HOST || 'localhost',
    port: parseInt(process.env.JAEGER_PORT || '6831'),
    serviceName: 'ctrl-alt-news-server',
  });

  const tracerProvider = new BasicTracerProvider();
  
  // Console exporter for dev debugging
  if (process.env.NODE_ENV === 'development') {
    tracerProvider.addSpanProcessor(new SimpleSpanProcessor(new ConsoleSpanExporter()));
  }

  // Batch exporter for production (sends to Jaeger)
  tracerProvider.addSpanProcessor(new BatchSpanProcessor(jaegerExporter));

  // Configure sampler: 10% in prod, 100% in dev
  const sampler = process.env.NODE_ENV === 'development'
    ? { shouldSample: () => true }
    : { shouldSample: () => Math.random() < 0.1 };

  const sdk = new NodeSDK({
    traceExporter: jaegerExporter,
    instrumentations: [registerInstrumentations()],
    sampler,
  });

  sdk.start();
  console.log('Tracing initialized');

  return () => sdk.shutdown();
}

export const tracer = require('@opentelemetry/api').trace.getTracer('ctrl-alt-news');
```

### HTTP Middleware Instrumentation

```typescript
// server/middleware/tracingMiddleware.ts
import { tracer } from '../tracing';
import { context, trace } from '@opentelemetry/api';

export function tracingMiddleware(req, res, next) {
  const span = tracer.startSpan(`HTTP ${req.method} ${req.path}`, {
    attributes: {
      'http.method': req.method,
      'http.url': req.url,
      'http.target': req.path,
      'http.host': req.hostname,
    },
  });

  // Generate or propagate request ID
  const requestId = req.headers['x-request-id'] || generateId();
  span.setAttributes({
    'http.request_id': requestId,
    'http.user_id': req.user?.id || 'anonymous',
  });

  // Run handler in span context
  return context.with(trace.setSpan(context.active(), span), () => {
    res.on('finish', () => {
      span.setAttributes({
        'http.status_code': res.statusCode,
        'http.response_content_length': res.get('content-length') || 0,
      });
      span.end();
    });

    next();
  });
}
```

### Database Query Instrumentation

```typescript
// server/lib/db.ts
import { tracer } from '../tracing';

export async function query(sql: string, params?: any[]) {
  const span = tracer.startSpan('database.query', {
    attributes: {
      'db.system': 'postgresql',
      'db.statement': sql,
      'db.operation': sql.split(/\s+/)[0].toUpperCase(), // SELECT, INSERT, etc.
      'db.params_count': params?.length || 0,
    },
  });

  try {
    const start = Date.now();
    const result = await pool.query(sql, params);
    
    span.setAttributes({
      'db.rows_affected': result.rowCount,
      'db.duration_ms': Date.now() - start,
    });

    return result;
  } catch (error) {
    span.recordException(error);
    throw error;
  } finally {
    span.end();
  }
}
```

### Trace Reading Guide (Markdown)

```markdown
# Understanding Request Traces

## Reading a Trace Waterfall

A trace shows the full lifecycle of a request:

```
GET /api/articles (500ms total)
├── parse_request (10ms)
├── auth_middleware (20ms)
├── db_query: SELECT articles (300ms)
│   └── parse_result (10ms)
├── cache_lookup: get_categories (30ms)
├── external_api: recommendation_service (100ms)
└── serialize_response (30ms)
```

Each bar represents a span. Horizontal alignment shows parallelism.

## Identifying Bottlenecks

1. **Find slowest request:** Jaeger UI → Search by latency (sort: slowest)
2. **Expand waterfall:** Click on request to see all spans
3. **Identify slowest span:** Look for widest bars (longest duration)
4. **Investigate:** Click span to see attributes (query, status, etc.)

## Common Slow Patterns

- **Wide database span:** Slow query or large result set → check explain plan
- **Multiple narrow database spans:** N+1 query problem → batch queries
- **Slow external API span:** Rate limiting or network latency → cache or timeout
- **Unaccounted time:** Missing instrumentation → add span for that component
```
