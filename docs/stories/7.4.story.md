## Technical Details

### GitHub Actions Workflow Structure

```yaml
name: CI/CD Pipeline

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  # Lint job
  lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-node@v3
        with:
          node-version: '18'
      - run: npm ci
      - run: npm run lint
      - run: npm run check  # TypeScript

  # Test job
  test:
    runs-on: ubuntu-latest
    needs: lint
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-node@v3
        with:
          node-version: '18'
      - run: npm ci
      - run: npm test

  # Build job
  build:
    runs-on: ubuntu-latest
    needs: [lint, test]
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-node@v3
        with:
          node-version: '18'
      - run: npm ci
      - run: npm run build
      - uses: actions/upload-artifact@v3
        with:
          name: build
          path: dist/

  # Deploy to staging
  deploy-staging:
    runs-on: ubuntu-latest
    needs: build
    if: github.ref == 'refs/heads/main' && github.event_name == 'push'
    environment:
      name: staging
      url: https://staging.example.com
    steps:
      - uses: actions/download-artifact@v3
        with:
          name: build
      - name: Deploy to staging
        run: |
          # Deploy script (SSH, scp, or cloud provider CLI)
          ./.github/scripts/deploy-staging.sh
      - name: Health check
        run: |
          curl -f https://staging.example.com/health || exit 1

  # Deploy to production (manual trigger)
  deploy-production:
    runs-on: ubuntu-latest
    needs: build
    if: github.event_name == 'workflow_dispatch'
    environment:
      name: production
      url: https://example.com
    steps:
      - uses: actions/download-artifact@v3
        with:
          name: build
      - name: Deploy to production
        run: |
          # Deploy script
          ./.github/scripts/deploy-production.sh
      - name: Health check
        run: |
          curl -f https://example.com/health || exit 1
      - name: Notify Slack
        run: |
          # Slack notification script
          ./.github/scripts/notify-slack.sh "✅ Production deployment successful"
```

### Health Check Endpoint

```typescript
// server/routes/health.ts
export async function getHealth(req: Request, res: Response) {
  try {
    // Check database connectivity
    await db.query('SELECT 1');
    
    // Check Redis connectivity (if configured)
    if (redis) {
      await redis.ping();
    }
    
    res.json({
      status: 'healthy',
      timestamp: new Date(),
      version: process.env.APP_VERSION,
      uptime: process.uptime()
    });
  } catch (error) {
    res.status(503).json({
      status: 'unhealthy',
      error: error.message
    });
  }
}
```

### Rollback Strategy

```bash
#!/bin/bash
# .github/scripts/rollback-production.sh

CURRENT_VERSION=$(curl -s https://api.example.com/version)
PREVIOUS_VERSION=$(cat /data/deployment-history | tail -2 | head -1)

if [ "$PREVIOUS_VERSION" != "" ]; then
  echo "Rolling back from $CURRENT_VERSION to $PREVIOUS_VERSION"
  # Download and deploy previous version
  ./deploy-production.sh $PREVIOUS_VERSION
  
  # Verify health
  curl -f https://example.com/health
else
  echo "No previous version available for rollback"
  exit 1
fi
```

## Risks & Mitigations

| Risk | Impact | Mitigation |
|------|--------|-----------|
| Deployment failure | Production outage | Health checks, rollback capability, manual approval for prod |
| Secrets exposure | Security breach | Use GitHub Secrets, never commit secrets, audit access |
| Database migration issue | Data corruption | Test migrations in staging first, rollback scripts |
| Failed health check | Bad deployment goes to prod | Automated rollback on health check failure |
| Long deployment time | Extended maintenance window | Optimize build/deploy, consider canary deployments (Phase 2) |
