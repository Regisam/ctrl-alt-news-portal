### Playwright Configuration Example

```typescript
// playwright.config.ts
import { defineConfig, devices } from '@playwright/test';

export default defineConfig({
  testDir: './e2e',
  fullyParallel: true,
  forbidOnly: !!process.env.CI,
  retries: process.env.CI ? 2 : 0,
  workers: process.env.CI ? 1 : undefined,
  timeout: 30000,
  expect: { timeout: 5000 },

  use: {
    baseURL: 'http://localhost:3000',
    trace: 'on-first-retry',
    screenshot: 'only-on-failure',
    video: 'retain-on-failure',
  },

  projects: [
    { name: 'chromium', use: { ...devices['Desktop Chrome'] } },
    { name: 'firefox', use: { ...devices['Desktop Firefox'] } },
```

### Page Object Pattern

```typescript
// e2e/fixtures/base.page.ts
import { Page } from '@playwright/test';

export class BasePage {
  constructor(protected page: Page) {}

  async goto(path: string) {
    await this.page.goto(path);
  }

  async getPageTitle() {
    return await this.page.title();
  }

  async waitForNavigation() {
    await this.page.waitForNavigation();
  }

  async close() {
    await this.page.close();
  }
}

// e2e/fixtures/home.page.ts
import { BasePage } from './base.page';

export class HomePage extends BasePage {
  async goto() {
    await super.goto('/');
    await this.page.waitForLoadState('networkidle');
  }

  async clickArticle(articleIndex: number) {
    const articles = this.page.locator('[data-testid="article-card"]');
    await articles.nth(articleIndex).click();
  }

  async searchArticles(query: string) {
    await this.page.fill('[data-testid="search-input"]', query);
    await this.page.press('[data-testid="search-input"]', 'Enter');
  }

  async selectCategory(category: string) {
    await this.page.click(`[data-testid="category-${category.toLowerCase()}"]`);
  }

  async isArticleVisible(title: string) {
    return await this.page.locator(`text=${title}`).isVisible();
  }
}
```

### User Journey Test Example

```typescript
// e2e/article.spec.ts
import { test, expect } from '@playwright/test';
import { HomePage } from './fixtures/home.page';
import { ArticlePage } from './fixtures/article.page';

test.describe('Article Reading Flow', () => {
  let homePage: HomePage;
  let articlePage: ArticlePage;

  test.beforeEach(async ({ page }) => {
    homePage = new HomePage(page);
    articlePage = new ArticlePage(page);
    await homePage.goto();
  });

  test('should read article from home page', async ({ page }) => {
    // Click first article
    await homePage.clickArticle(0);

    // Wait for article page
    await articlePage.waitForArticleContent();

    // Verify article title visible
    const title = await articlePage.getArticleTitle();
    expect(title.length).toBeGreaterThan(0);

    // Verify content loaded
    const content = await articlePage.getArticleContent();
    expect(content).toBeTruthy();
  });

  test('should navigate between articles', async ({ page }) => {
    // Click first article
    await homePage.clickArticle(0);
    const title1 = await articlePage.getArticleTitle();

    // Navigate to next article
    await articlePage.clickNextArticle();
    const title2 = await articlePage.getArticleTitle();

    // Titles should be different
    expect(title1).not.toBe(title2);
  });

  test('should return to home after reading', async ({ page }) => {
    // Go to article
    await homePage.clickArticle(0);

    // Click back button
    await articlePage.clickBackButton();

    // Should be back on home
    expect(await homePage.isArticleListVisible()).toBeTruthy();
  });
});
```
